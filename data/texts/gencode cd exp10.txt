%%
[a-z]        { yylval.dval = yytext[0]; return LETTER; }
[0-9]        { yylval.dval = yytext[0]; return NUM; }
[\t]         { return 0; }
.            { return yytext[0]; }
%%

int yywrap() {
    return 1;
}








%{
#include <stdio.h>
#include <string.h>

char *gencode(char op, char operand1, char operand2);
int k = 1;
%}

%union {
    char dval;
}

%token <dval> NUM LETTER
%type <dval> E

%left '+' '-'
%left '*' '/' '%'

%%
statement : LETTER '=' E { printf("%c = t%d\n", $1, k-1); }
          ;

E : E '+' E   { $$ = *gencode('+', $1, $3); }
  | E '-' E   { $$ = *gencode('-', $1, $3); }
  | E '*' E   { $$ = *gencode('*', $1, $3); }
  | E '/' E   { $$ = *gencode('/', $1, $3); }
  | E '%' E   { $$ = *gencode('%', $1, $3); }
  | '(' E ')' { $$ = $2; }
  | NUM       { $$ = $1; }
  | LETTER    { $$ = $1; }
  ;
%%

char *gencode(char op, char operand1, char operand2) {
    static char buffer[100];
    sprintf(buffer, "t%d = %c %c %c\n", k++, operand1, op, operand2);
    printf("%s", buffer);
    return buffer;
}
